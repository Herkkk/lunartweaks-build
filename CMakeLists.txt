cmake_minimum_required(VERSION 3.24)
project(LunarTweaks VERSION 2.0.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Target Windows x64, or use build-mingw.sh with a cross compiler")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(LunarTweaks WIN32 app/main.cpp app/catalog.cpp app/resources.rc)
target_compile_definitions(LunarTweaks PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_include_directories(LunarTweaks PRIVATE app)
set_property(SOURCE app/resources.rc PROPERTY INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/app")
target_link_libraries(LunarTweaks PRIVATE gdiplus dwmapi comctl32 winhttp bcrypt ncrypt crypt32 shell32 ole32 uuid powrprof iphlpapi ws2_32 advapi32 gdi32 user32 msimg32 winmm setupapi)
if(MSVC)
  target_compile_options(LunarTweaks PRIVATE /W4 /permissive- /utf-8 /sdl /guard:cf)
  target_link_options(LunarTweaks PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
else()
  target_compile_options(LunarTweaks PRIVATE -O2 -Wall -Wextra -Werror)
  target_link_options(LunarTweaks PRIVATE -municode -static -static-libgcc -static-libstdc++ -s -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
endif()
add_custom_command(TARGET LunarTweaks POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:LunarTweaks>/assets"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/LunarTweaks.ini" "$<TARGET_FILE_DIR:LunarTweaks>/LunarTweaks.ini"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/docs/CUSTOMER-START.txt" "$<TARGET_FILE_DIR:LunarTweaks>/START-HERE.txt"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/docs/THIRD-PARTY-NOTICES.txt" "$<TARGET_FILE_DIR:LunarTweaks>/THIRD-PARTY-NOTICES.txt")

add_executable(LunarTweaksOwner WIN32 app/owner_start.cpp app/owner_start.rc)
set_target_properties(LunarTweaksOwner PROPERTIES OUTPUT_NAME "OPEN-LUNARTWEAKS")
set_property(SOURCE app/owner_start.rc PROPERTY INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/app")
target_link_libraries(LunarTweaksOwner PRIVATE user32)
if(MSVC)
  target_compile_options(LunarTweaksOwner PRIVATE /W4 /permissive- /utf-8 /sdl /guard:cf)
  target_link_options(LunarTweaksOwner PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
else()
  target_compile_options(LunarTweaksOwner PRIVATE -O2 -Wall -Wextra -Werror)
  target_link_options(LunarTweaksOwner PRIVATE -municode -static -static-libgcc -static-libstdc++ -s -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
endif()
